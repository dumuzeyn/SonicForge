cmake_minimum_required(VERSION 3.24)
project(SonicForge VERSION 3.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
file(GLOB_RECURSE SF_BUILD_SOURCES CONFIGURE_DEPENDS RELATIVE "${CMAKE_SOURCE_DIR}"
  "native/*.cpp" "native/*.h" "native/*.inc" "native/*.in" "native/resources/*.json" "native/*.rc" "scripts/*.ps1" "packaging/*.iss" "packaging/*.cpp" "packaging/*.rc" "packaging/*.manifest")
list(APPEND SF_BUILD_SOURCES "CMakeLists.txt" "native/dependencies.json" "native/runtime-lock.json")
list(SORT SF_BUILD_SOURCES)
set(SF_SOURCE_RECORD "")
foreach(SF_FILE IN LISTS SF_BUILD_SOURCES)
  if(EXISTS "${CMAKE_SOURCE_DIR}/${SF_FILE}")
    file(SHA256 "${CMAKE_SOURCE_DIR}/${SF_FILE}" SF_FILE_HASH)
    string(APPEND SF_SOURCE_RECORD "${SF_FILE}:${SF_FILE_HASH}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/${SF_FILE}")
  endif()
endforeach()
string(SHA256 SF_SOURCE_DIGEST "${SF_SOURCE_RECORD}")
file(WRITE "${CMAKE_BINARY_DIR}/source-files.sha256" "${SF_SOURCE_RECORD}")
configure_file(native/build_info.h.in generated/build_info.h @ONLY)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia Concurrent Network Svg Test)
set(WHISPER_SOURCE "${CMAKE_SOURCE_DIR}/build/native-deps/whisper" CACHE PATH "whisper.cpp source")
set(DEMUCS_SOURCE "${CMAKE_SOURCE_DIR}/build/native-deps/demucs" CACHE PATH "demucs.cpp source")
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Prepare pinned whisper.cpp sources with scripts/build_native.ps1")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)
if(MINGW)
  target_compile_options(ggml-cpu PRIVATE -include "${CMAKE_SOURCE_DIR}/native/compat_windows.h")
endif()
if(NOT EXISTS "${DEMUCS_SOURCE}/src/model.hpp")
  message(FATAL_ERROR "Prepare pinned demucs.cpp sources with scripts/build_native.ps1")
endif()
find_package(OpenMP REQUIRED)
file(GLOB DEMUCS_FILES CONFIGURE_DEPENDS "${DEMUCS_SOURCE}/src/*.cpp")
if(WIN32)
  # Upstream's narrow fopen cannot open models in non-ASCII Windows paths.
  file(READ "${DEMUCS_SOURCE}/src/model_load.cpp" DEMUCS_LOADER)
  string(REPLACE "fopen(model_file.c_str(), \"rb\")" "_wfopen(std::filesystem::u8path(model_file).c_str(), L\"rb\")" DEMUCS_LOADER "${DEMUCS_LOADER}")
  set(DEMUCS_PREVIOUS_LOADER "")
  if(EXISTS "${CMAKE_BINARY_DIR}/model_load_utf8.cpp")
    file(READ "${CMAKE_BINARY_DIR}/model_load_utf8.cpp" DEMUCS_PREVIOUS_LOADER)
  endif()
  if(NOT DEMUCS_PREVIOUS_LOADER STREQUAL DEMUCS_LOADER)
    file(WRITE "${CMAKE_BINARY_DIR}/model_load_utf8.cpp" "${DEMUCS_LOADER}")
  endif()
  list(REMOVE_ITEM DEMUCS_FILES "${DEMUCS_SOURCE}/src/model_load.cpp")
  list(APPEND DEMUCS_FILES "${CMAKE_BINARY_DIR}/model_load_utf8.cpp")
endif()
add_library(sonic_demucs STATIC ${DEMUCS_FILES})
target_include_directories(sonic_demucs PUBLIC "${DEMUCS_SOURCE}/src" "${DEMUCS_SOURCE}/vendor/eigen")
target_link_libraries(sonic_demucs PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(sonic_demucs PUBLIC EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
if(MINGW)
  target_compile_options(sonic_demucs PRIVATE -O2 -g0 -w -fno-unsafe-math-optimizations)
endif()
add_library(sonic_core STATIC
  native/core.cpp native/reliability.cpp native/audio_changes.cpp native/metadata.cpp native/covers.cpp native/cover_texture.cpp native/cover_symbols.cpp native/neural.cpp native/alignment.cpp native/icons.cpp native/transliteration.cpp native/process_session.cpp)
target_include_directories(sonic_core PUBLIC native)
target_link_libraries(sonic_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Svg PRIVATE whisper sonic_demucs)
target_compile_definitions(sonic_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
qt_add_resources(sonic_core "resources" PREFIX "/" BASE "native/resources" FILES
  native/resources/ru.json native/resources/en.json native/resources/en-native.json native/resources/en-reliability.json
  native/resources/models.json native/resources/transliteration.json native/resources/ctc-ru.json native/resources/ctc-en.json)
qt_add_resources(sonic_core "help_resource" PREFIX "/" BASE "native/resources" FILES native/resources/help-en.html native/resources/chevron-down.svg native/resources/chevron-up.svg)
qt_add_resources(sonic_core "brand_resource" PREFIX "/" BASE "." FILES assets/sonic_forge_mark.ico)
qt_add_resources(sonic_core "check_resource" PREFIX "/" BASE "native/resources" FILES native/resources/check.svg native/resources/context-help.json)
qt_add_executable(SonicForge WIN32 native/main.cpp native/window.cpp native/window_project.cpp native/window_audio.cpp native/window_cover.cpp native/timeline.cpp native/window.h native/timeline.h native/app.rc)
if(WIN32)
  target_link_libraries(SonicForge PRIVATE shell32 user32 ole32 gdi32 uuid)
endif()
target_include_directories(SonicForge PRIVATE "${CMAKE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(SonicForge PRIVATE sonic_core Qt6::Widgets Qt6::Multimedia Qt6::Concurrent)
if(MINGW)
  target_compile_options(sonic_core PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
  target_compile_options(SonicForge PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
qt_add_executable(sonic_tests native/tests.cpp)
target_link_libraries(sonic_tests PRIVATE sonic_core Qt6::Test)
if(WIN32)
  target_link_libraries(sonic_tests PRIVATE shell32 ole32 gdi32 uuid)
endif()
add_test(NAME native_core COMMAND sonic_tests)
qt_add_executable(reliability_tests native/reliability_tests.cpp)
set_target_properties(sonic_tests reliability_tests PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(reliability_tests PRIVATE sonic_core Qt6::Test)
add_test(NAME reliability_core COMMAND reliability_tests)
set_tests_properties(reliability_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 180)
set_tests_properties(native_core PROPERTIES TIMEOUT 180)
add_test(NAME reliability_gui COMMAND SonicForge --reliability-smoke --result "${CMAKE_SOURCE_DIR}/validation/reliability-gui.json")
set_tests_properties(reliability_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 90)
set_tests_properties(native_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
install(TARGETS SonicForge RUNTIME DESTINATION .)
