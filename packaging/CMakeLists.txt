cmake_minimum_required(VERSION 3.24)
project(SonicForgePackaging LANGUAGES CXX RC)
add_executable(SonicForge-Uninstall WIN32 uninstall_launcher.cpp uninstall_launcher.rc)
target_compile_features(SonicForge-Uninstall PRIVATE cxx_std_17)
target_include_directories(SonicForge-Uninstall PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(SonicForge-Uninstall PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(SonicForge-Uninstall PRIVATE advapi32 shell32 user32)
if(MSVC)
  # The application manifest is embedded explicitly by uninstall_launcher.rc.
  target_link_options(SonicForge-Uninstall PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_compile_options(SonicForge-Uninstall PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(SonicForge-Uninstall PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
